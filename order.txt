# Astribot S1 PI0.5 training and evaluation commands

Use this dataset for both versions:
repo_id: local/cup_selection_s1_action
root: /home/jchen572/code_file/RBM_lerobot/lerobotv3_datasets/cup_selection_s1_action
task: Select the red cup and place it on the black coaster

Both PI0.5 runs below are full-parameter finetuning.


1. Start the Astribot image service

cd /home/jchen572/code_file/astribot_service
conda activate teleimage
export ASTRIBOT_SDK_ROOT=/home/jchen572/code_file/astribot_service/astribot_sdk_aarch64
export ASTRIBOT_LOG=1
source /opt/ros/humble/setup.bash
python -m astribot_image_service.server \
    --sdk-root /home/jchen572/code_file/astribot_service/astribot_sdk_aarch64 \
    --bind-host 0.0.0.0 \
    --sdk-freq 30 \
    --head-port 6100 \
    --left-wrist-port 6101 \
    --right-wrist-port 6102 \
    --jpeg-quality 80 \
    --enable-control \
    --execute-control \
    --control-bind-host 0.0.0.0 \
    --control-port 6200 \
    --control-frequency 250 \
    --control-action-frequency 120 \
    --control-way filter


2. Optional image client preview

cd /home/jchen572/Code_file/RBM_lerobot
conda activate lerobot
export PYTHONPATH=/home/jchen572/Code_file/RBM_lerobot/astribot_lerobot/lerobot/src
python -m astribot_image_service.client \
    --host 192.168.0.11 \
    --head-port 6100 \
    --left-wrist-port 6101 \
    --right-wrist-port 6102


3. Real robot dataset action replay

This uses keyboard gates: type c then Enter to connect, press r to move to the dataset first frame, press s to start replay, press q to shutdown.

cd /home/jchen572/code_file/RBM_lerobot
conda activate astribot_lerobot
export PYTHONPATH=/home/jchen572/code_file/RBM_lerobot/astribot_lerobot/lerobot/src
python astribot_lerobot/eval_robot/replay_robot.py \
    --repo_id=local/cup_selection_s1_action \
    --root=/home/jchen572/code_file/RBM_lerobot/lerobotv3_datasets/red_cup_selection_left_s1 \
    --task="Select the red cup and place it on the black coaster" \
    --episodes=0 \
    --frequency=30 \
    --arm=Astribot_S1 \
    --ee='' \
    --visualization=false \
    --send_real_robot=true \
    --image_transport=teleimager \
    --image_server_address=192.168.0.11 \
    --head_camera_zmq_port=6100 \
    --left_wrist_camera_zmq_port=6101 \
    --right_wrist_camera_zmq_port=6102 \
    --head_camera_image_height=720 \
    --head_camera_image_width=1280 \
    --wrist_camera_image_height=360 \
    --wrist_camera_image_width=640 \
    --use_wrist_cameras=true \
    --astribot_control_transport=remote \
    --astribot_control_server_address=192.168.0.11 \
    --astribot_control_port=6200 \
    --astribot_control_timeout_s=5 \
    --astribot_remote_interpolation_steps=4 \
    --camera_features='["observation.images.head","observation.images.left","observation.images.right"]' \
    --astribot_control_way=filter \
    --astribot_use_wbc=false \
    --ready_lift_m=0 \
    --ready_max_joint_delta=0.25 \
    --arm_low_level_velocity_limit=8.0 \
    --arm_tracking_error_limit=0.35 \
    --dataset_start_tolerance=0.08 \
    --slow_move_tracking_wait_timeout_s=8 \
    --init_timeout_s=60


4. PI0.5 25D full-parameter finetune

cd /home/jchen572/Code_file/RBM_lerobot
conda activate astribot_lerobot
export PYTHONPATH=/home/jchen572/Code_file/RBM_lerobot/astribot_lerobot/lerobot/src
CUDA_VISIBLE_DEVICES=1 python astribot_lerobot/lerobot/src/lerobot/scripts/lerobot_train.py \
    --dataset.repo_id=local/cup_selection_s1_action \
    --dataset.root=/home/jchen572/Code_file/RBM_lerobot/lerobotv3_datasets/red_cup_selection_left_s1 \
    --dataset.feature_names='["action","observation.state","observation.images.head","observation.images.left","observation.images.right"]' \
    --dataset.astribot_action_view=full \
    --policy.type=pi05 \
    --policy.pretrained_path=lerobot/pi05_base \
    --policy.compile_model=true \
    --policy.gradient_checkpointing=true \
    --policy.dtype=bfloat16 \
    --policy.device=cuda \
    --policy.push_to_hub=false \
    --output_dir=/home/jchen572/Code_file/RBM_lerobot/astribot_lerobot/lerobot/outputs/pi05_astribot_s1_25d \
    --job_name=pi05_astribot_s1_25d \
    --batch_size=16 \
    --steps=40000 \
    --save_freq=20000 \
    --wandb.enable=true \
    --wandb.disable_artifact=true
