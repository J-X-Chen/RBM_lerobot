# Astribot S1 / LeRobot v3 commands
# Run these from a bash shell on the robot/training machine.

SCRIPT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]:-$0}")" && pwd)
ROOT=${ROOT:-$SCRIPT_DIR}
PROJECT_DIR=${PROJECT_DIR:-$ROOT/astribot_lerobot}
DATASET=${DATASET:-$ROOT/lerobotv3_datasets/lerobotv3_test_joints}
REPO_ID=${REPO_ID:-local/lerobotv3_test_joints}
TASK=${TASK:-"Replay the Astribot S1 joint test trajectory."}
FEATURES='["action","observation.state","observation.images.head","observation.images.left","observation.images.right"]'
CAMERA_FEATURES='["observation.images.head","observation.images.left","observation.images.right"]'
export PYTHONPATH=$PROJECT_DIR/lerobot/src:${PYTHONPATH:-}


################################################################################
# 0. Dataset sanity check
################################################################################

cd $ROOT

python -c 'import json, pathlib; root=pathlib.Path("'$DATASET'"); info=json.loads((root/"meta"/"info.json").read_text()); print("dataset:", root); print("robot_type:", info.get("robot_type")); print("fps:", info.get("fps")); print("frames:", info.get("total_frames")); print("features:", sorted(info["features"].keys())); print("state_shape:", info["features"]["observation.state"]["shape"]); print("action_shape:", info["features"]["action"]["shape"])'


################################################################################
# 1. PI0.5 smoke train first
################################################################################

cd $PROJECT_DIR/lerobot

CUDA_VISIBLE_DEVICES=1 python src/lerobot/scripts/lerobot_train.py \
  --dataset.repo_id=$REPO_ID \
  --dataset.root=$DATASET \
  --dataset.feature_names="$FEATURES" \
  --dataset.compress_dex3_actions=auto \
  --dataset.image_transforms.enable=true \
  --policy.type=pi05 \
  --policy.pretrained_path=lerobot/pi05_base \
  --policy.compile_model=false \
  --policy.gradient_checkpointing=true \
  --policy.dtype=bfloat16 \
  --policy.device=cuda \
  --policy.push_to_hub=false \
  --output_dir=./outputs/pi05_astribot_s1_smoke \
  --job_name=pi05_astribot_s1_smoke \
  --wandb.enable=false \
  --wandb.disable_artifact=true \
