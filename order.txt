# 6. Start Astribot S1 image bridge, keep this terminal running
################################################################################

# From the workstation, copy S1-side code/SDK to Orin with same-name replacement:
#   python send_code_to_S1.py ./astribot_image_service 192.168.0.11 /home/jchen572/code_file/astribot_service
#   python send_code_to_S1.py ./astribot_sdk_aarch64 192.168.0.11 /home/jchen572/code_file/astribot_service

cd $ROOT
# On S1/Orin, use the native aarch64 SDK. Do not copy the workstation/x86
# astribot_sdk_ros2 directory to the robot. If your SDK lives elsewhere, set
# ASTRIBOT_SDK_ROOT before this block.
# The aarch64 SDK native extensions are built against NumPy 1.x. Keep Python
# 3.10 on S1, but make sure this environment has the SDK deps:
#   sudo apt install -y ros-humble-tf-transformations
#   python -m pip uninstall -y meta
#   python -m pip install "numpy==1.24.0" filterpy transforms3d tabulate h5py opencv-python==4.10.0.82
conda activate teleimage
export ASTRIBOT_SDK_ROOT=./astribot_sdk_aarch64
export ASTRIBOT_LOG=1
source /opt/ros/humble/setup.bash
source "$ASTRIBOT_SDK_ROOT/env.sh"

python -m astribot_image_service.server \
  --sdk-root "$ASTRIBOT_SDK_ROOT" \
  --bind-host 0.0.0.0 \
  --head-port=55555 \
  --left-wrist-port=55556 \
  --right-wrist-port=55557 \
  --high-control-rights \
  --enable-control \
  --execute-control \
  --control-bind-host=0.0.0.0 \
  --control-port=6200 \
  --control-frequency=250 \
  --control-action-frequency=30 \
  --control-way=filter

# Optional preview: open another terminal after the image bridge is running.
# If cv2 is missing, install the viewer extra: pip install -e ".[viewer]"
cd $ROOT
python -m astribot_image_service.client \
  --host 127.0.0.1 \
  --head-port=55555 \
  --left-wrist-port=55556 \
  --right-wrist-port=55557
