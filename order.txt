# Astribot S1 PI0.5 training and evaluation commands

Use this dataset for both versions:
repo_id: local/cup_selection_s1_action
root: /home/jchen572/code_file/RBM_lerobot/lerobotv3_datasets/cup_selection_s1_action
task: Select the red cup and place it on the black coaster

Both PI0.5 runs below are full-parameter finetuning.


1. Start the Astribot image service

cd /home/jchen572/Code_file/RBM_lerobot
conda activate teleimage
export PYTHONPATH=/home/jchen572/Code_file/RBM_lerobot/astribot_lerobot/lerobot/src
export ASTRIBOT_SDK_ROOT=/home/jchen572/Code_file/RBM_lerobot/astribot_sdk_aarch64
export ASTRIBOT_LOG=1
source /opt/ros/humble/setup.bash
source /home/jchen572/Code_file/RBM_lerobot/astribot_sdk_aarch64/env.sh
python -m astribot_image_service.server \
    --sdk-root /home/jchen572/Code_file/RBM_lerobot/astribot_sdk_aarch64 \
    --bind-host 0.0.0.0 \
    --bind-port 6100 \
    --frequency 30 \
    --jpeg-quality 80 \
    --camera 1:front \
    --camera 2:left_wrist \
    --camera 3:right_wrist


2. Optional image client preview

cd /home/jchen572/Code_file/RBM_lerobot
conda activate lerobot
export PYTHONPATH=/home/jchen572/Code_file/RBM_lerobot/astribot_lerobot/lerobot/src

    --server tcp://192.168.0.11:6100 \

3. PI0.5 25D full-parameter finetune

cd /home/jchen572/Code_file/RBM_lerobot
conda activate astribot_lerobot
export PYTHONPATH=/home/jchen572/Code_file/RBM_lerobot/astribot_lerobot/lerobot/src
CUDA_VISIBLE_DEVICES=1 python astribot_lerobot/lerobot/src/lerobot/scripts/lerobot_train.py \
    --dataset.repo_id=local/cup_selection_s1_action \
    --dataset.root=/home/jchen572/Code_file/RBM_lerobot/lerobotv3_datasets/red_cup_selection_left_s1 \
    --dataset.feature_names='["action","observation.state","observation.images.head","observation.images.left","observation.images.right"]' \
    --dataset.astribot_action_view=full \
    --policy.type=pi05 \
    --policy.pretrained_path=lerobot/pi05_base \
    --policy.compile_model=true \
    --policy.gradient_checkpointing=true \
    --policy.dtype=bfloat16 \
    --policy.device=cuda \
    --policy.push_to_hub=false \
    --output_dir=/home/jchen572/Code_file/RBM_lerobot/astribot_lerobot/lerobot/outputs/pi05_astribot_s1_25d \
    --job_name=pi05_astribot_s1_25d \
    --batch_size=16 \
    --steps=40000 \
    --save_freq=20000 \
    --wandb.enable=true \
    --wandb.disable_artifact=true
